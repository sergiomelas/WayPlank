# ------------------------------------------------------------------------------
# 1. MUTTER 50+ (Minimal Compositor Only - Standalone 800x600 Window)
# ------------------------------------------------------------------------------

# Installation:
sudo apt install --no-install-recommends mutter mutter-dev-bin

# Test Method A (Recommended: Two Terminals):
# Terminal 1 - Start persistent nested Mutter 800x600 window:
MUTTER_DEBUG_DUMMY_MODE_SPECS=800x600 env -u LD_PRELOAD GTK_CSD=1 mutter --wayland --devkit --no-x11 --virtual-monitor=800x600 --wayland-display=wayland-9

# Terminal 2 - Launch Wayplank inside that Mutter window (loads standard dock1):
WAYLAND_DISPLAY=wayland-9 env -u LD_PRELOAD GTK_CSD=1 /home/sergio/Others/WayPlank/0.4_Wayland/build/wayplank -r -d

# Test Method B (Single-Command One-Liner):
MUTTER_DEBUG_DUMMY_MODE_SPECS=800x600 env -u LD_PRELOAD GTK_CSD=1 mutter --wayland --devkit --no-x11 --virtual-monitor=800x600 --wayland-display=wayland-9 -- sh -c "env -u LD_PRELOAD GTK_CSD=1 /home/sergio/Others/WayPlank/0.4_Wayland/build/wayplank -r -d"

#For screene shot use:
WAYLAND_DISPLAY=wayland-0 spectacle -b -f -o /home/sergio/Others/WayPlank/0.4_Wayland/screenshot.png

# Purge:
sudo apt purge mutter mutter-dev-bin
sudo apt autoremove --purge


# ------------------------------------------------------------------------------
# 2. GNOME SHELL & MUTTER (Full GNOME Desktop with Extensions Support)
# ------------------------------------------------------------------------------

# Minimal Installation (Debian Sid GNOME 50+):
sudo apt install --no-install-recommends \
    gnome-shell \
    gnome-shell-extension-prefs \
    gnome-terminal

# Optional: Add full session if you want to log in directly from SDDM/GDM:
# sudo apt install --no-install-recommends gnome-session

# Test Method (Runs GNOME Shell 50+ inside a 1280x720 nested window on host Wayland):
# Terminal 1 - Start nested GNOME Shell:
MUTTER_DEBUG_DUMMY_MODE_SPECS=1280x720 dbus-run-session gnome-shell --wayland --no-x11 --virtual-monitor=1280x720 --wayland-display=wayland-9

# Terminal 2 - Launch Wayplank inside that GNOME nested window:
WAYLAND_DISPLAY=wayland-9 dbus-run-session /home/sergio/Others/WayPlank/0.4_Wayland/build/wayplank -d

# Managing Wayplank GNOME Shell Extension:
# Extensions directory: ~/.local/share/gnome-shell/extensions/
# List installed extensions:
# gnome-extensions list
# Enable extension:
# gnome-extensions enable wayplank-bridge@wayplank.org

# Purge:
sudo apt purge \
    gnome-shell \
    gnome-shell-extension-prefs \
    gnome-terminal
sudo apt autoremove --purge
